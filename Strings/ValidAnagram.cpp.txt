class Solution {
public:
    bool isAnagram(string s, string t) {
        int n = s.length();
        int arr[26] = {0};
        if(s.length()!=t.length()){
            return false;
        }
        for(int i=0;i<n;i++){
            arr[s[i]-'a']++;
            arr[t[i]-'a']--;
        }
        for(int i=0;i<26;i++){
            if(arr[i]>0){
                return false;
            }
        }
        return true;
    }
};
